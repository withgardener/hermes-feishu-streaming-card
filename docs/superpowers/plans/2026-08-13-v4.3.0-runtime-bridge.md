# V4.3.0 Runtime Bridge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Route authenticated Feishu Gateway turns through Hermes official plugin hooks while preserving the required Hybrid patch bridge, exact native-delivery behavior, legacy compatibility, and fail-open operation.

**Architecture:** `hermes_plugin.py` starts as the inert wheel entry point produced by the Integration Core plan, then becomes the fail-open production bootstrap in this milestone; its runtime facade delegates to `hermes_plugin_runtime.py`. That module owns ingress/turn state, deterministic identities, a bounded observer worker, signed Sidecar transport and all official-hook normalization. The sidecar remains the sole owner of event validation, card session state, rendering, Feishu delivery, and a bounded event-id fence; `hook_runtime.py` exposes only narrow Hybrid patch bridge functions and does not regain lifecycle ownership.

**Tech Stack:** Python 3.9+, asyncio, threading, dataclasses, bounded `queue.Queue`, hashlib/canonical JSON, aiohttp, setuptools entry points, pytest and pytest-asyncio.

  class TurnState(str, Enum):
      PENDING_START = "pending-start"
      CARD_ACTIVE = "card-active"
      NATIVE_BYPASS = "native-bypass"
      TERMINAL = "terminal"


  @dataclass(frozen=True)
  class IngressBinding:
      profile_id: str
      session_id: str
      gateway_session_key: str
      generation: str
      chat_id: str
      incoming_message_id: str
      reply_to_message_id: str
      thread_id: str
      expires_at: float


  @dataclass
  class TurnBinding:
      ingress: IngressBinding
      turn_id: str
      state: TurnState = TurnState.PENDING_START

      @property
      def accepts_observer_events(self) -> bool:
          return self.state is TurnState.CARD_ACTIVE

      def record_started_result(self, result: object) -> TurnState:
          if self.state is not TurnState.PENDING_START:
              return self.state
          if _is_explicit_started_acceptance(result):
              self.state = TurnState.CARD_ACTIVE
          else:
              self.state = TurnState.NATIVE_BYPASS
          return self.state

      def finish(self) -> bool:
          if self.state is TurnState.TERMINAL:
              return False
          self.state = TurnState.TERMINAL
          return True

  ```python
  @dataclass(frozen=True)
  class ObserverEvent:
      sequence: int
      producer: str
      payload: dict[str, object]


  class TurnEventCoordinator:
      def __init__(self, turn_id: str, *, max_pending: int = 64,
                   deliver: Callable[[ObserverEvent], None] | None = None,
                   start_worker: bool = True) -> None:
          self.turn_id = turn_id
          self._next = 0
          self._barrier: int | None = None
          self._closed = False
          self._lock = threading.Lock()
          self._queue: queue.Queue[ObserverEvent] = queue.Queue(maxsize=max_pending)
          self._deliver = deliver or (lambda item: None)

      def next_sequence(self, producer: str) -> int:
          if producer not in {"plugin", "patch", "legacy-patch"}:
              raise ValueError("unknown producer")
          with self._lock:
              value, self._next = self._next, self._next + 1
              return value

      def close_terminal_barrier(self) -> int:
          with self._lock:
              self._barrier = self._next - 1
              return self._barrier

      def next_terminal_sequence(self) -> int:
          with self._lock:
              if self._barrier is None:
                  raise ValueError("terminal barrier is not closed")
              value, self._next = self._next, self._next + 1
              self._closed = True
              return value

      def event_id(self, kind: str, *, item_id: str = "", phase: str = "") -> str:
          if kind in {"started", "completed", "failed"}:
              return f"turn:{self.turn_id}:{kind}"
          if kind not in {"tool", "approval", "subagent"} or not item_id:
              raise ValueError("invalid event identity")
          if phase not in {"started", "terminal"}:
              raise ValueError("invalid event phase")
          return f"{kind}:{self.turn_id}:{item_id}:{phase}"

      def submit_observer(self, payload: dict[str, object], *, producer: str) -> bool:
          with self._lock:
              if self._closed or self._barrier is not None:
                  return False
              if producer not in {"plugin", "patch", "legacy-patch"}:
                  raise ValueError("unknown producer")
              sequence, self._next = self._next, self._next + 1
              event = ObserverEvent(sequence, producer, dict(payload, sequence=sequence))
              try:
                  self._queue.put_nowait(event)
              except queue.Full:
                  return False
              return True

      def drain_before_terminal(self, timeout_seconds: float) -> None:
          deadline = time.monotonic() + max(0.0, timeout_seconds)
          while self._queue.unfinished_tasks and time.monotonic() < deadline:
              time.sleep(0.001)

      def close(self) -> None:
          with self._lock:
              self._closed = True
 ```

Allocate the sequence before queue insertion. `drain_before_terminal()` waits only until the bounded timeout, then closes queue admission and does not wait indefinitely. `close()` makes late callbacks return `False`; it never blocks a Hermes callback.
